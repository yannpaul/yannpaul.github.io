    package main

import (
	"flag"
	"fmt"
	"io/fs"
	"os"
	"path/filepath"

type Config struct {
	Command string
	Path    string
}

func getConfig() (Config, error) {
	if len(os.Args) < 2 {
		return Config{}, fmt.Errorf("expected a subcommand: find, du, or perm")
	}

	switch os.Args[1] {
	case "find":
		fs := flag.NewFlagSet("find", flag.ExitOnError)
		path := fs.String("path", ".", "path to search")
		fs.Parse(os.Args[2:])
		return Config{Command: "find", Path: *path}, nil
	case "du":
		fs := flag.NewFlagSet("du", flag.ExitOnError)
		path := fs.String("path", ".", "path to inspect")
		fs.Parse(os.Args[2:])
		return Config{Command: "du", Path: *path}, nil
	case "perm":
		fs := flag.NewFlagSet("perm", flag.ExitOnError)
		path := fs.String("path", ".", "path to inspect")
		fs.Parse(os.Args[2:])
		return Config{Command: "perm", Path: *path}, nil
	default:
		return Config{}, fmt.Errorf("unknown subcommand %q", os.Args[1])
	}
}

func walk(root string) error {
	return filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		fmt.Println(path)
		return nil
	})
}

func run(cfg Config) error {
	switch cfg.Command {
	case "find":
		return walk(cfg.Path)
	case "du":
		fmt.Println("burrow du: not implemented yet")
		return nil
	case "perm":
		fmt.Println("burrow perm: not implemented yet")
		return nil
	}
	return nil
}

func main() {
	cfg, err := getConfig()
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	if err := run(cfg); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}

New: find dispatches to its own subcommand

find's current case in getConfig() handles its own -path flag directly. Today that case stops doing the work itself and instead becomes a dispatcher, exactly like getConfig() itself is — it looks at os.Args[2] (the argument right after find) to decide which kind of find, then hands the rest (os.Args[3:]) to a flag set of its own. Only one kind exists so far: name, which matches filenames against a glob pattern using filepath.Match(pattern, name) — the same shell-style wildcards you'd use at a terminal (*, ?, [abc]), matched against one path element at a time, not a whole path.

  • Notice the pattern: this is identical in shape to what getConfig() itself does — check a length, switch on the next argument, build a flag set from one slice further into os.Args. Nested subcommands are just this same recipe applied again, one argument deeper.

    Stuck? Show the function 
    func getFindConfig() (Config, error) {
	if len(os.Args) < 3 {
		return Config{}, fmt.Errorf("expected a find subcommand: name")
	}

	switch os.Args[2] {
	case "name":
		fs := flag.NewFlagSet("find name", flag.ExitOnError)
		pattern := fs.String("pattern", "*", "glob pattern to match filenames against")
		path := fs.String("path", ".", "path to search")
		fs.Parse(os.Args[3:])
		return Config{Command: "find name", Path: *path, Pattern: *pattern}, nil
	default:
		return Config{}, fmt.Errorf("unknown find subcommand %q", os.Args[2])
	}
}

  • filepath.Match itself can return an error (a malformed pattern) — don't ignore it, return it the same way you return the walk's own err.

    Stuck? Show the function 
    func findName(root, pattern string) error {
	return filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		matched, err := filepath.Match(pattern, d.Name())
		if err != nil {
			return err
		}
		if matched {
			fmt.Println(path)
		}
		return nil
	})
}

Idiom

filepath.Match's patterns are shell globs, not regular expressions — * matches any run of characters, ? matches exactly one, [abc] matches one of a set — and critically, * does not cross a path separator, and Match is only ever called here against d.Name(), one filename at a time, never the full accumulated path. That's why matching still needs filepath.WalkDir doing the recursion — Match itself has no idea about directory trees at all, it just compares one name against one pattern.

You've now built the mission's exact example from scratch: burrow find name --pattern "*.go" is foo bar baz --arg val, three levels deep, using nothing but the standard library. Lesson 6 applies this same nested-dispatch recipe to du instead of find — different data, identical shape.
